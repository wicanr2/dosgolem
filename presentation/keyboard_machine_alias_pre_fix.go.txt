package presentation

import (
	"testing"

	"github.com/wicanr2/dosgolem/host"
	"github.com/wicanr2/dosgolem/internal/dos"
	"github.com/wicanr2/dosgolem/internal/machine"
)

// Disposable Issue #18 / spec233 counterexample. The public DOS.M alias can
// change after a valid bridge is constructed; production does not revalidate.
func TestDraftBIOSBridgePublicMachineAliasCanRedirectKey(t *testing.T) {
	first, second := machine.New(), machine.New()
	panel, err := host.NewPanelController(host.OutputScale2)
	if err != nil {
		t.Fatal(err)
	}
	bios := dos.New(first, ".")
	bridge, err := NewKeyboardBridgeWithBIOS(panel, first, bios)
	if err != nil {
		t.Fatal(err)
	}
	bios.M = second
	_, route, err := bridge.DeliverBIOSKey(dos.Key{Scan: 0x1c, ASCII: 0x0d})
	if err != nil || !route.ForwardToDOS || first.BIOSKeyCount() != 0 || second.BIOSKeyCount() != 1 {
		t.Fatalf("machine alias negative not reproduced: route=%+v err=%v first=%d second=%d",
			route, err, first.BIOSKeyCount(), second.BIOSKeyCount())
	}
}
